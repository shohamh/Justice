## Task 11 – Issue 12: Exemption requests not reaching commander approval queue

**Context:** Two separate problems compound here:

1. **Scope query never fires**: `GET /exemption-requests/pending` gates on `scope_root_ids(session, user)`. That function returns nodes where `HierarchyNode.commander_id == user.id`. If `set_commander` was never called (or had a bug), `commander_id` is NULL and the function returns an empty set → the route short-circuits with `[]`. Task 10 (fix `set_commander`) is a prerequisite.

2. **Holding-node blind spot**: During registration, the soldier is placed in `holding_node` (not the target node). Exemption requests submitted during registration are attached to the soldier while they're in the holding node. The holding node has no commander, so those requests are invisible in every commander's queue. After enrollment approval (`soldier.hierarchy_node_id = requested_node_id`), those pending requests DO become visible — but the commander has no way of knowing to go check the exemptions tab after approving enrollment. They need a notification AND the requests need to be surfaced directly in the enrollment approval UI.

- [ ] **Step 2: Create the export router**

Create `backend/app/routes/export.py`:

```python
from __future__ import annotations
import io
import uuid
from fastapi import APIRouter, Depends, Query
from fastapi.responses import StreamingResponse
from sqlalchemy import select
from sqlalchemy.orm import Session
from openpyxl import Workbook
from app.auth.deps import require_password_changed
from app.db.models import HierarchyNode, Soldier, SoldierExemption, PersonalConstraint, ExemptionType
from app.db.session import get_session

router = APIRouter(prefix="/export", tags=["export"])


@router.get("/soldiers")
def export_soldiers(
    node_id: uuid.UUID = Query(...),
    session: Session = Depends(get_session),
    user: Soldier = Depends(require_password_changed),
) -> StreamingResponse:
    # Find all nodes in subtree
    all_nodes = session.execute(
        select(HierarchyNode).where(HierarchyNode.path_ids.any(node_id))
    ).scalars().all()
    node_ids = {n.id for n in all_nodes}

    soldiers = session.execute(
        select(Soldier).where(
            Soldier.hierarchy_node_id.in_(node_ids),
            Soldier.left_at.is_(None),
        )
    ).scalars().all()

    wb = Workbook()
    ws = wb.active
    ws.title = "חיילים"
    headers = ["מספר אישי", "שם", "דרגה", "קצין", "בוגר בה\"ד 1", "מגדר", "גיוס", "סיום חובה", "שחרור", "טלפון"]
    ws.append(headers)
    for s in soldiers:
        ws.append([
            s.personal_number, s.full_name, s.rank or "", str(s.is_officer or ""), str(s.bahad1_graduate),
            s.gender or "", str(s.enlistment_date or ""), str(s.mandatory_end_date or ""),
            str(s.discharge_date or ""), s.phone or "",
        ])

    buf = io.BytesIO()
    wb.save(buf)
    buf.seek(0)
    return StreamingResponse(
        buf,
        media_type="application/vnd.openxmlformats-officedocument.spreadsheetml.sheet",
        headers={"Content-Disposition": "attachment; filename=soldiers.xlsx"},
    )


@router.get("/import-template")
def export_import_template(
    session: Session = Depends(get_session),
    user: Soldier = Depends(require_password_changed),
) -> StreamingResponse:
    """Return an Excel in the exact format expected by the /import page."""
    wb = Workbook()
    ws = wb.active
    ws.title = "ייבוא חיילים"
    # Match exactly the columns the ImportPage expects
    # (Check ImportPage.tsx to get exact column names)
    headers = ["מספר אישי", "שם מלא", "דרגה", "קצין (כן/לא)", "בוגר בה\"ד 1 (כן/לא)", "מגדר", "תאריך גיוס", "סיום חובה", "שחרור", "טלפון", "אימייל", "מסגרת (שם)"]
    ws.append(headers)
    # Example row
    ws.append(["1234567", "ישראל ישראלי", "סמל", "לא", "לא", "זכר", "2022-01-01", "2025-01-01", "", "050-0000000", "", "יחידה א"])
    buf = io.BytesIO()
    wb.save(buf)
    buf.seek(0)
    return StreamingResponse(
        buf,
        media_type="application/vnd.openxmlformats-officedocument.spreadsheetml.sheet",
        headers={"Content-Disposition": "attachment; filename=import-template.xlsx"},
    )
```

**Note:** Before implementing, read `frontend/src/pages/ImportPage.tsx` to get the exact column names expected.

- [ ] **Step 4: Create ExportPage.tsx**

Create `frontend/src/pages/planning/ExportPage.tsx`:

```tsx
import { useState, useEffect } from "react";
import { useTranslation } from "react-i18next";
import Layout from "../../components/Layout";
import { listNodes } from "../../api/hierarchy";
import { NodeDTO } from "../../api/hierarchy";

export default function ExportPage() {
  const { t } = useTranslation();
  const [nodes, setNodes] = useState<NodeDTO[]>([]);
  const [selectedNodeId, setSelectedNodeId] = useState("");

  useEffect(() => {
    listNodes().then(setNodes).catch(() => {});
  }, []);

  function downloadSoldiers() {
    if (!selectedNodeId) return;
    window.location.href = `/api/export/soldiers?node_id=${selectedNodeId}`;
  }

  function downloadImportTemplate() {
    window.location.href = `/api/export/import-template`;
  }

  return (
    <Layout>
      <div className="space-y-6" dir="rtl">
        {/* Panel 1: Soldiers export */}
        <section className="bg-white dark:bg-gray-800 rounded-lg shadow p-6 space-y-4">
          <h2 className="text-xl font-semibold">ייצוא פוטנציאל חיילים</h2>
          <p className="text-sm text-gray-500 dark:text-gray-400">
            ייצוא Excel עם כל החיילים ביחידה נבחרת, פרטיהם ופטוריהם.
          </p>
          <div>
            <label className="block text-sm font-medium mb-1">בחר יחידה</label>
            <select
              className="border rounded p-2 text-sm dark:bg-gray-700 dark:border-gray-600 dark:text-gray-100"
              value={selectedNodeId}
              onChange={e => setSelectedNodeId(e.target.value)}
            >
              <option value="">— בחר —</option>
              {nodes.map(n => (
                <option key={n.id} value={n.id}>{n.name}</option>
              ))}
            </select>
          </div>
          <button
            disabled={!selectedNodeId}
            onClick={downloadSoldiers}
            className="bg-green-600 text-white px-4 py-2 rounded text-sm disabled:opacity-50"
          >
            ⬇ הורד Excel חיילים
          </button>
        </section>

        {/* Panel 2: Import template */}
        <section className="bg-white dark:bg-gray-800 rounded-lg shadow p-6 space-y-4">
          <h2 className="text-xl font-semibold">תבנית ייבוא</h2>
          <p className="text-sm text-gray-500 dark:text-gray-400">
            הורד קובץ Excel בפורמט ייבוא – מוכן למילוי ויצירת חיילים חדשים.
          </p>
          <button
            onClick={downloadImportTemplate}
            className="bg-indigo-600 text-white px-4 py-2 rounded text-sm"
          >
            ⬇ הורד תבנית ייבוא
          </button>
        </section>
      </div>
    </Layout>
  );
}
```
